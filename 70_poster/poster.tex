\subsection{Research Questions} \vspace{-2em}

\begin{itemize}
  \item \textbf{RQ1:} How well can deep learning detect oil slicks in Sentinel-1 SAR imagery, and how does performance change between an in-distribution random split and a geographic out-of-distribution split (Mediterranean)?
  \item \textbf{RQ2:} Can a Geospatial Foundation Model improve over a CNN baseline, and what do XAI methods reveal about the features that drive the predictions?
\end{itemize}

\section{Methodology}
\subsection{Overall Setup} \vspace{-2em}
\begin{itemize}
  \item \textbf{CNN baseline} trained from scratch on OilSlick
  \item \textbf{Terramind} \cite{Jakubik_2025_ICCV} trained while freezing the backbone and only training a classification head on OilSlick
  \item \textbf{Grid search} performed for both models (36 total training runs)
  \item \textbf{Find best model} (based on val loss) for each model and split type (4 total) for individual final test set evaluation.
\end{itemize}

\subsection{Crucial Steps before Training} \vspace{-2em}
\begin{itemize}
  \item \textbf{Data analysis} $\to$ understand data distributions and apply imagery filtering.
  \item \textbf{Data filtering} was crucial for this project, by finding corrupt TIFF files and imagery with >= 40\% of nodata in it.

\end{itemize}
