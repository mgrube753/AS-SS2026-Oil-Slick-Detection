\section{Introduction} \vspace{-1em}
\subsection{Project Motivation} \vspace{-1.5em}

\begin{itemize}
  \item \textbf{Oil slicks} $\to$ severe damage \& hard to monitor manually
  \item \textbf{Sentinel-1 SAR imagery} $\to$ day / night / all-weather monitoring possible
  \item \textbf{Short surface waves} $\to$ slicks as dark patches in SAR imagery
\end{itemize}
\vspace{.5em}
\subsection{Research Questions} \vspace{-1.5em}
\begin{itemize}
  \item \textbf{RQ1:} How well can deep learning detect oil slicks in \textit{Sentinel-1 SAR imagery}, and how does performance change between an in-distribution random split and a geographic out-of-distribution split (\textit{Mediterranean})?
  \item \textbf{RQ2:} Can a \textit{Geospatial Foundation Model} improve over a CNN baseline, and what do XAI methods reveal about the features that drive the predictions?
\end{itemize}

\section{Methodology} \vspace{.5em}
\subsection{Experimental Setup} \vspace{-1.5em}
\begin{itemize}
  \item \textbf{CNN baseline} $\to$ trained from scratch on OilSlick
  \item \textbf{Terramind} \cite{11445875} $\to$ frozen backbone, only training a classification head
  \item \textbf{Grid search} $\to$ performed for both models (36 total training runs)
  \item \textbf{Find best configuration} $\to$ based on val loss for each model \& split
  \item \textbf{Final evaluation} $\to$ 4 final models $\to$ individual test set
\end{itemize}
\vspace{.85em}
\subsection{Crucial Steps before Training} \vspace{-1.5em}
\begin{itemize}
  \item \textbf{Data analysis} $\to$ understand data distributions \& apply imagery filtering
  \item \textbf{Data filtering} $\to$ find corrupt TIFF files \& imagery with >= 40\% of nodata
\end{itemize}

\section{Conclusion \& Future Work}
\begin{itemize}
  \item \textbf{RQ1:} Oil Slick F1 scores $\sim$\textbf{0.71--0.77}, only mild OOD drop ($\sim$\textbf{0.03--0.04}) \vspace{.3em} % surprisingly robust across regions (see F1 score column for Oil Slick (geo cnn vs random cnn / geo gfm vs random gfm)). so, ML/DL can be used to detect oil slicks in SAR imagery across several regions
  \item \textbf{RQ2:} GFM $\to$ higher Slick F1 ($\uparrow$ \textbf{0.03}), lower No-Slick F1 ($\downarrow$ \textbf{0.07}) \vspace{.2em}\\$\to$ \textit{no clear improvement over CNN} $\mid$ XAI not applied (time) \vspace{.3em} % see GFM random and CNN random; GFM geo and CNN geo even differ less
  \item \textbf{Limitations:} All models trade precision for recall (Oil Slick) % recall > precision for each config; random CNN is more balanced and shows highest Slick precision with 0.7285, but lowest Slick recall with 0.7586 --> F1=0.7432
   $\to$ \textit{False alarms} on lookalikes as key failure mode $\mid$ Dataset quality \textit{(invalid samples)} \vspace{.3em} % even after filtering <= -50db imagery / >= 40% nodata / std=0.0 imagery (corrupt)
  \item \textbf{Future Work:} XAI-driven failure analysis to \textit{interpret} model behavior on lookalikes, and experimenting on (larger) SAR datasets with less invalid samples % since we are no domain-specific experts, we cannot tell why invalid samples still remain in the dataset after filtering
\end{itemize}
