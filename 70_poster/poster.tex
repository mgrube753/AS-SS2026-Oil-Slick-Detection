\section{Introduction} \vspace{-1em}
\subsection{Project Motivation} \vspace{-2em}

\begin{itemize}
  \item \textbf{Oil slicks} $\to$ severe damage \& hard to monitor manually
  \item \textbf{Sentinel-1 SAR imagery} $\to$ day / night / all-weather monitoring possible
  \item \textbf{Short surface waves} $\to$ slicks as dark patches in SAR imagery
\end{itemize}

\subsection{Research Questions} \vspace{-2em}

  \textbf{RQ1:} How well can deep learning detect oil slicks in Sentinel-1 SAR imagery, and how does performance change between an in-distribution random split and a geographic out-of-distribution split (Mediterranean)? \vspace{1em} \\
  \textbf{RQ2:} Can a Geospatial Foundation Model improve over a CNN baseline, and what do XAI methods reveal about the features that drive the predictions?

\section{Methodology}
\subsection{Experimental Setup} \vspace{-2em}
\begin{itemize}
  \item \textbf{CNN baseline} $\to$ trained from scratch on OilSlick
  \item \textbf{Terramind} \cite{11445875} $\to$ frozen backbone, only training a classification head
  \item \textbf{Grid search} $\to$ performed for both models (36 total training runs)
  \item \textbf{Find best configuration} $\to$ based on val loss for each model \& split
  \item \textbf{Final evaluation} $\to$ 4 final models $\to$ individual test set
\end{itemize}

\subsection{Crucial Steps before Training} \vspace{-2em}
\begin{itemize}
  \item \textbf{Data analysis} $\to$ understand data distributions \& apply imagery filtering
  \item \textbf{Data filtering} $\to$ find corrupt TIFF files \& imagery with >= 40\% of nodata
\end{itemize}

\section{Conclusion \& Future Work}
\begin{itemize}
  \item \textbf{RQ1:} Oil Slick F1 score $\sim$0.71--0.77 $\mid$ only mild OOD drop ($\sim$0.03 F1) \vspace{.25em} % surprisingly robust across regions (see F1 score column for Oil Slick (geo cnn vs random cnn / geo gfm vs random gfm))
  \item \textbf{RQ2:} GFM $\to$ higher Slick F1 ($\uparrow$ 0.03) $\mid$ lower No-Slick F1 ($\downarrow$ 0.07) \\$\to$ no clear improvement over CNN; XAI not applied (time) \vspace{.25em}
  \item \textbf{Limitations:} All models trade precision for recall (Slick) $\to$ false alarms on lookalikes as key failure mode $\mid$ Dataset quality (invalid samples) \vspace{.25em} % even after filtering <= -50db imagery / >= 40% nodata / std=0.0 imagery (corrupt)
  \item \textbf{Future Work:} XAI-driven failure analysis to understand model behavior on lookalikes $\mid$ Larger SAR datasets with less invalid samples
\end{itemize}
